# boards/cosmac/cdp1802/sim1802/scripts/banks.txt
#
# Code placement for the HWB profile (tools/cosmac/cosmac_bank.py,
# ADR 0013).  One rule per line, first match wins:
#
#   <bank>|auto|fixed   <archive-glob>:<member-glob>
#
# Objects that match no rule stay in the 32 KiB fixed ROM, as do libgcc and
# the board library (not given to the tool).  From the architecture
# library, everything that runs while the bank state or a register frame
# is being changed must stay fixed: the reset code's helpers, NCRT, the
# far-call trampoline and the per-thread bank switch, the interrupt entry
# and dispatch, the context switch, task exit and the signal trampoline
# (a frame resumes it directly, not through a call).  The rest of the
# architecture library and the memory manager are ordinary C code and go
# to banks: all read-only data stays in fixed ROM, which is the scarce
# part (the far call costs about 400 machine cycles, ADR 0013).

fixed   libarch.a:cosmac_ncrt.o
fixed   libarch.a:cosmac_farcall.o
fixed   libarch.a:cosmac_banking.o
fixed   libarch.a:cosmac_irqentry.o
fixed   libarch.a:cosmac_irqsave.o
fixed   libarch.a:cosmac_doirq.o
fixed   libarch.a:cosmac_doswitch.o
fixed   libarch.a:cosmac_switchcontext.o
fixed   libarch.a:cosmac_exit.o
fixed   libarch.a:cosmac_sigdeliver.o

# Small functions called from everywhere: each call from a bank to another
# costs about 330 machine cycles of trampolines (NSH's ps made 3 394 far
# calls, 636 of them to memcpy).  They stay in fixed ROM while it has room,
# most frequently called first; ostest, whose read-only strings fill the
# fixed ROM, banks them instead.

# The timer tick runs 50 times a second: every far call on its path costs
# about 400 cycles (measured at Step 07: a tick with this path banked took
# 5 795 cycles, more than the 5 000-cycle tick period at 100 Hz).

fixed   libsched.a:irq_dispatch.o
fixed   libsched.a:sched_processtick.o
fixed   libsched.a:wd_start.o

# Console output crosses many objects for every character (formatting,
# the stdio stream, the buffer), and so does formatting into memory
# (snprintf, procfs files).  Keep that chain in one bank and the
# write path (file write, serial driver, syslog) in another, so that far
# calls happen per line, not per character (Step 09: one character cost
# about 13 000 machine cycles with the chain spread over five banks).
