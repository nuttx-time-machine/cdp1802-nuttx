# boards/cosmac/cdp1802/sim1802/scripts/banks.txt
#
# Code placement for the HWB profile (tools/cosmac/cosmac_bank.py,
# ADR 0013).  One rule per line, first match wins:
#
#   <bank>|auto|fixed   <archive-glob>:<member-glob>
#
# Objects that match no rule stay in the 32 KiB fixed ROM, as do libgcc and
# the board library (not given to the tool).  The architecture library holds
# the reset code, the NCRT and far-call routines and the interrupt and
# context-switch code: it must stay fixed, except the 64-bit division
# helpers (plain C functions, not used by the tick or the switch path).  Every far call costs about 400
# machine cycles more than a near call, so frequently called code (the
# memory manager) stays fixed as well.

auto    libarch.a:cosmac_div64.o
fixed   libarch.a:*
fixed   libmm.a:*

# The timer tick runs 50 times a second: every far call on its path costs
# about 400 cycles (measured at Step 07: a tick with this path banked took
# 5 795 cycles, more than the 5 000-cycle tick period at 100 Hz).

fixed   libsched.a:irq_dispatch.o
fixed   libsched.a:sched_processtick.o
fixed   libsched.a:wd_start.o
fixed   libsched.a:clock_sched_ticks.o
fixed   libsched.a:clock_systime_ticks.o
fixed   libsched.a:sched_switchcontext.o
fixed   libsched.a:sched_roundrobin.o

auto    libsched.a:*
auto    libc.a:*
auto    libfs.a:*
auto    libdrivers.a:*
auto    libapps.a:*
